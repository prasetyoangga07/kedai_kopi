LOAD DATA INFILE 'transaction_details.csv'
INTO TABLE transaction_details
FIELDS TERMINATED BY ','
OPTIONALLY ENCLOSED BY '"'
LINES TERMINATED BY '\r\n'
IGNORE 1 LINES
(
    transaction_id,
    variant_id,
    qty,
    unit_price,
    created_at,
    @updated_at
)
SET 
    updated_at = NULLIF(@updated_at, '');
